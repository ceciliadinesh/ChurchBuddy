function getList(data) {
    return Array.isArray(data) ? data : data?.items || [];
}

function getLectorName(lector, lectorId) {
    if (!lector) {
        return lectorId != null ? `Lector #${lectorId}` : "Unassigned";
    }

    const firstName = lector.first_name || lector.firstname || "";
    const lastName = lector.last_name || lector.lastname || "";
    const combined = `${firstName} ${lastName}`.trim();

    return (
        lector.full_name ||
        lector.name ||
        combined ||
        lector.email ||
        `Lector #${lectorId}`
    );
}

function getAssignmentName(item) {
    return (
        item.special_text ||
        item.lector_name ||
        getLectorName(item.lector, item.lector_id)
    );
}

function getMassTime(item) {
    return (
        item.mass_slot?.mass_time ||
        item.mass_slot?.time ||
        item.mass_time ||
        item.slot_time ||
        ""
    );
}

function parseMassTime(value) {
    const raw = String(value || "").trim().toUpperCase();
    const match = raw.match(/(\d{1,2}):(\d{2})(?::\d{2})?\s*(AM|PM)?/);

    if (!match) return null;

    let hour = Number(match[1]);
    const minute = Number(match[2]);

    if (match[3] === "PM" && hour < 12) hour += 12;
    if (match[3] === "AM" && hour === 12) hour = 0;

    return { hour, minute };
}

function getMassMinutes(item) {
    const t = parseMassTime(getMassTime(item));
    return t ? t.hour * 60 + t.minute : 0;
}

function formatMassTime(value) {
    if (!value) return "Time unavailable";

    const t = parseMassTime(value);
    if (!t) return value;

    const period = t.hour >= 12 ? "PM" : "AM";
    const displayHour = t.hour % 12 || 12;

    return `${displayHour}:${String(t.minute).padStart(2, "0")} ${period}`;
}

function getReadingLabel(item) {
    if (item.responsibility === "Reading" && item.position_number) {
        return `Rdng.${item.position_number}`;
    }

    const value = String(
        item.reading ||
        item.reading_type ||
        item.responsibility ||
        item.position_name ||
        item.position ||
        ""
    ).trim();

    if (!value) return "";

    const normalized = value.toLowerCase().replace(/\s+/g, "");

    const aliases = {
        ang: "Ang.",
        angelus: "Ang.",
        "rdng.1": "Rdng.1",
        reading1: "Rdng.1",
        "rdng.2": "Rdng.2",
        reading2: "Rdng.2",
        ps: "P.S.",
        "p.s.": "P.S.",
        pof: "POF",
    };

    return aliases[normalized] || value;
}

function formatRosterDate(dateValue) {
    if (!dateValue) return { date: "", day: "" };

    const date = new Date(`${String(dateValue).slice(0, 10)}T12:00:00`);

    if (Number.isNaN(date.getTime())) {
        return { date: String(dateValue), day: "" };
    }

    return {
        date: date.toLocaleDateString("en-GB", {
            day: "2-digit",
            month: "short",
        }).toUpperCase(),
        day: date.toLocaleDateString("en-US", {
            weekday: "short",
        }).toUpperCase(),
    };
}

function escapeHtml(value) {
    return String(value ?? "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}

function getDateOnly(value) {
    return String(value || "").slice(0, 10);
}

// Time chosen when the date was frozen (defaults to 7:00 AM for old records)
function getFrozenTime(item) {
    return formatMassTime(item?.frozen_time || "7:00 AM");
}

// All frozen date+time entries for one calendar date
function getFrozenForDate(frozenDates, dateStr) {
    return frozenDates.filter(
        (item) => getDateOnly(item.frozen_date) === dateStr
    );
}

// Is this exact date + Mass time frozen?
function isSlotFrozen(frozenDates, dateStr, timeLabel) {
    return frozenDates.some(
        (item) =>
            getDateOnly(item.frozen_date) === dateStr &&
            getFrozenTime(item) === timeLabel
    );
}

// Sunday note helpers

function getSundaySlotDate(slot) {
    return getDateOnly(
        slot.slot_date ||
        slot.date ||
        slot.assignment_date
    );
}

function getSundaySlotTime(slot) {
    return formatMassTime(
        slot.mass_time ||
        slot.mass_slot?.mass_time ||
        slot.mass_slot?.time ||
        slot.slot_time ||
        ""
    );
}

function isSundaySlot(slot, time) {
    return (
        getSundaySlotTime(slot) === time &&
        Boolean(getSundaySlotDate(slot))
    );
}

async function copyText(text) {
    try {
        await navigator.clipboard.writeText(text);
    } catch {
        const textarea = document.createElement("textarea");
        textarea.value = text;
        textarea.style.position = "fixed";
        textarea.style.opacity = "0";

        document.body.appendChild(textarea);
        textarea.select();
        document.execCommand("copy");
        textarea.remove();
    }
}

function getRosterSlotIndex(item) {
    const t = parseMassTime(getMassTime(item));
    if (!t) return -1;

    const key = `${t.hour}:${String(t.minute).padStart(2, "0")}`;

    const map = {
        "7:00": 0,
        "7:30": 1,
        "9:45": 2,
        "18:00": 3,
    };

    return key in map ? map[key] : -1;
}

function buildRosterGroups(assignments, sundaySlots = [], frozenDates = []) {
    const byDate = new Map();

    // Normal assignments
    assignments.forEach((item) => {
        const date = getDateOnly(item.assignment_date || item.date);
        if (!date) return;

        if (!byDate.has(date)) byDate.set(date, []);
        byDate.get(date).push(item);
    });

    // Include Sunday note dates
    sundaySlots.forEach((slot) => {
        const date = getSundaySlotDate(slot);
        if (date && !byDate.has(date)) byDate.set(date, []);
    });

    // Include frozen dates
    frozenDates.forEach((item) => {
        const date = getDateOnly(item.frozen_date);
        if (date && !byDate.has(date)) byDate.set(date, []);
    });

    return [...byDate.entries()]
        .sort(([a], [b]) => a.localeCompare(b))
        .map(([dateValue, items]) => {
            const slotItems = [[], [], [], []];
            const readingLabels = new Set();

            items.forEach((item) => {
                const label = getReadingLabel(item);
                if (label) readingLabels.add(label);

                const slot = getRosterSlotIndex(item);

                if (slot >= 0) {
                    slotItems[slot].push(item);
                }
            });

            const labels = [...readingLabels].sort((a, b) => {
                const ai = READING_ORDER.indexOf(a);
                const bi = READING_ORDER.indexOf(b);

                if (ai === -1 && bi === -1) return a.localeCompare(b);
                if (ai === -1) return 1;
                if (bi === -1) return -1;

                return ai - bi;
            });

            const rows = labels.map((label) => ({
                reading: label,
                slots: slotItems.map((slot) =>
                    slot
                        .filter(
                            (item) =>
                                getReadingLabel(item) === label &&
                                (
                                    item.lector_id != null ||
                                    Boolean(item.special_text)
                                )
                        )
                        .map((item) => ({
                            text: getAssignmentName(item),
                            special: Boolean(item.special_text),
                        }))
                ),
            }));

            // Sunday 7:30 AM and 9:45 AM notes go into the existing Rdng.1 row
            const dateSundaySlots = sundaySlots.filter(
                (slot) => getSundaySlotDate(slot) === dateValue
            );

            dateSundaySlots.forEach((slot) => {
                const note = String(slot.note || "").trim();
                if (!note) return;

                const time = getSundaySlotTime(slot);
                const columnIndex = ROSTER_SLOTS.indexOf(time);

                if (columnIndex === -1) return;

                let readingRow = rows.find(
                    (row) => row.reading === "Rdng.1"
                );

                if (!readingRow) {
                    readingRow = {
                        reading: "Rdng.1",
                        slots: [[], [], [], []],
                    };

                    rows.push(readingRow);
                }

                readingRow.slots[columnIndex].push({
                    text: note,
                    special: true,
                });
            });

            // Frozen notes (one per frozen time) go into the matching
            // time column of an existing row
            const frozenForDate = getFrozenForDate(frozenDates, dateValue);

            frozenForDate.forEach((frozenDate) => {
                const frozenNote = frozenDate.reason?.trim();
                if (!frozenNote) return;

                const columnIndex = ROSTER_SLOTS.indexOf(
                    getFrozenTime(frozenDate)
                );

                if (columnIndex === -1) return;

                let readingRow =
                    rows.find((row) => row.reading === "Rdng.1") ||
                    rows[0];

                if (!readingRow) {
                    readingRow = {
                        reading: "Rdng.1",
                        slots: [[], [], [], []],
                    };

                    rows.push(readingRow);
                }

                readingRow.slots[columnIndex].push({
                    text: frozenNote,
                    special: true,
                });
            });

            // Keep one row for dates that have no assignments or notes
            if (rows.length === 0) {
                rows.push({
                    reading: "",
                    slots: [[], [], [], []],
                });
            }

            return {
                dateValue,
                ...formatRosterDate(dateValue),
                rows,
            };
        });
}

function buildCalendarWeeks(month, year, assignments, sundaySlots) {
    if (!month || !year) return [];

    const m = Number(month);
    const y = Number(year);
    const pad = (n) => String(n).padStart(2, "0");

    const byDate = new Map();
    const sundaySlotsByDate = new Map();

    assignments.forEach((item) => {
        const d = getDateOnly(item.assignment_date || item.date);
        if (!d) return;

        if (!byDate.has(d)) byDate.set(d, []);
        byDate.get(d).push(item);
    });

    sundaySlots.forEach((slot) => {
        const d = getSundaySlotDate(slot);
        if (!d) return;

        if (!sundaySlotsByDate.has(d)) {
            sundaySlotsByDate.set(d, []);
        }

        sundaySlotsByDate.get(d).push(slot);
    });

    byDate.forEach((list) =>
        list.sort((a, b) => {
            const diff = getMassMinutes(a) - getMassMinutes(b);
            if (diff !== 0) return diff;

            const ra = RESPONSIBILITY_ORDER[a.responsibility] || 99;
            const rb = RESPONSIBILITY_ORDER[b.responsibility] || 99;

            if (ra !== rb) return ra - rb;

            return (a.position_number || 0) - (b.position_number || 0);
        })
    );

    const offset = (new Date(y, m - 1, 1).getDay() + 6) % 7;
    const daysInMonth = new Date(y, m, 0).getDate();

    const cells = [
        ...Array(offset).fill(null),
        ...Array.from({ length: daysInMonth }, (_, i) => {
            const dateStr = `${y}-${pad(m)}-${pad(i + 1)}`;
            const date = new Date(`${dateStr}T12:00:00`);
            const isSunday = date.getDay() === 0;

            return {
                dateStr,
                dayNumber: i + 1,
                items: byDate.get(dateStr) || [],
                sundaySlots: isSunday
                    ? sundaySlotsByDate.get(dateStr) || []
                    : [],
            };
        }),
    ];

    while (cells.length % 7 !== 0) cells.push(null);

    const weeks = [];

    for (let i = 0; i < cells.length; i += 7) {
        weeks.push(cells.slice(i, i + 7));
    }

    return weeks;
}

    const [message, setMessage] = useState("");

    );

    );

    // ---------------- API ----------------

    function headers() {
        return {
            "Content-Type": "application/json",
            Authorization: `Bearer ${sessionStorage.getItem("admin_token")}`,
        };
    }

    async function request(path, options = {}) {
        const response = await fetch(`${API_URL}${path}`, {
            ...options,
            headers: { ...headers(), ...options.headers },
        });

        const data = await response.json().catch(() => ({}));

        if (!response.ok) {
            let detail = "";

            if (typeof data.detail === "string") {
                detail = data.detail;
            } else if (data.detail?.message) {
                detail = data.detail.message;

                if (Array.isArray(data.detail.warnings)) {
                    detail += ` ${data.detail.warnings.join(" ")}`;
                }
            } else if (Array.isArray(data.detail)) {
                detail = data.detail
                    .map((d) => d.msg || JSON.stringify(d))
                    .join(" ");
            }

            const requestError = new Error(
                detail || `Request failed (${response.status}).`
            );

            requestError.detail = data.detail;
            throw requestError;
        }

        return data;
    }

    // ---------------- LOAD ----------------

    async function loadAssignments(id) {
        if (!id) {
            setAssignments([]);
            return;
        }

        setAssignments(getList(await request(`/assignments/${id}`)));
    }

    async function loadFrozenDates(id) {
        if (!id) {
            setFrozenDates([]);
            return;
        }

        setFrozenDates(getList(await request(`/frozen-dates/${id}`)));
    }

    async function loadSundaySlots(id) {
        if (!id) {
            setSundaySlots([]);
            return;
        }

        setSundaySlots(getList(await request(`/sunday-slots/${id}`)));
    }

    // ---------------- SUNDAY NOTE EDITING ----------------

    function startEditingSundayNote(slot) {
        setSelectedSundaySlot(slot);
        setSundayNoteDraft(slot.note || "");
        setEditingSundayNote(true);
        setError("");
        setMessage("");
    }

    function cancelEditingSundayNote() {
        setSelectedSundaySlot(null);
        setSundayNoteDraft("");
        setEditingSundayNote(false);
    }

    async function saveSundayNote() {
        if (!selectedSundaySlot || !selectedId) return;

        const slotDate = getSundaySlotDate(selectedSundaySlot);
        const massSlotId = selectedSundaySlot.mass_slot_id;

        if (!slotDate || !massSlotId) {
            setError(
                "The Sunday Mass slot is missing its date or Mass slot ID. Check the Sunday slots API response."
            );
            return;
        }

        setWorking(true);
        setError("");
        setMessage("");

        try {
            await request(`/sunday-slots/${selectedId}`, {
                method: "PUT",
                body: JSON.stringify({
                    slot_date: slotDate,
                    mass_slot_id: massSlotId,
                    note: sundayNoteDraft.trim() || null,
                }),
            });

            await loadSundaySlots(selectedId);

            setEditingSundayNote(false);
            setSelectedSundaySlot(null);
            setSundayNoteDraft("");
            setMessage(
                `${getSundaySlotTime(selectedSundaySlot)} note saved successfully.`
            );
        } catch (err) {
            setError(err.message || "Could not save Sunday Mass note.");
        } finally {
            setWorking(false);
        }
    }

    // ---------------- FROZEN DATE NOTE EDITING ----------------

    function startEditingFrozenNote(item) {
        setEditingFrozenId(item.id);
        setFrozenNoteDraft(item.reason || "");
        setFrozenTimeDraft(getFrozenTime(item));
        setError("");
        setMessage("");
    }

    function cancelEditingFrozenNote() {
        setEditingFrozenId(null);
        setFrozenNoteDraft("");
        setFrozenTimeDraft("7:00 AM");
    }

    async function saveFrozenNote() {
        const frozenDate = frozenDates.find(
            (item) => item.id === editingFrozenId
        );

        if (!frozenDate) {
            setError("Could not find this frozen date.");
            return;
        }

        if (frozenNoteDraft.length > 100) {
            setError("Frozen date note cannot exceed 100 characters.");
            return;
        }

        // Changing the time must not collide with another frozen time
        // on the same date.
        const clash = frozenDates.some(
            (item) =>
                item.id !== frozenDate.id &&
                getDateOnly(item.frozen_date) ===
                getDateOnly(frozenDate.frozen_date) &&
                getFrozenTime(item) === formatMassTime(frozenTimeDraft)
        );

        if (clash) {
            setError("That date and time has already been frozen.");
            return;
        }

        setWorking(true);
        setError("");
        setMessage("");

        try {
            const result = await request(`/frozen-dates/${frozenDate.id}`, {
                method: "PATCH",
                body: JSON.stringify({
                    reason: frozenNoteDraft.trim() || null,
                    frozen_time: frozenTimeDraft,
                }),
            });

            setFrozenDates((prev) =>
                prev.map((item) =>
                    item.id === frozenDate.id
                        ? {
                            ...item,
                            reason:
                                result.reason ??
                                (frozenNoteDraft.trim() || null),
                            frozen_time:
                                result.frozen_time ?? frozenTimeDraft,
                        }
                        : item
                )
            );

            cancelEditingFrozenNote();
            setMessage("Frozen date note saved successfully.");

                        google_form_id: result.google_form_id,
                        google_form_url: result.form_url,

    // ---------------- ADD MONTH ----------------

    async function createMonth() {
        const month = Number(newMonth);
        const year = Number(newYear);

        if (!Number.isInteger(month) || month < 1 || month > 12) {
            setError("Please select a valid month.");
            return;
        }

        if (!Number.isInteger(year) || year < 2026) {
            setError("Year must be 2026 or later.");
            return;
        }

        const title = new Date(year, month - 1, 1).toLocaleDateString(
            "en-US",
            { month: "long", year: "numeric" }
        );

        setCreatingMonth(true);
        setError("");
        setMessage("");

        try {
            const result = await request("/monthly-forms/", {
                method: "POST",
                body: JSON.stringify({
                    month,
                    year,
                    title,
                    is_open: true,
                }),
            });

            const nextId = String(result.id);

            setForms((prev) =>
                [...prev, result].sort(
                    (a, b) => b.year - a.year || b.month - a.month
                )
            );

            setSelectedId(nextId);
            setAssignments([]);
            setFrozenDates([]);
            setSundaySlots([]);
            setDateInput("");
            setFreezeNoteInput("");
            setSelectedDay(null);
            cancelEditing();
            cancelEditingFrozenNote();
            cancelEditingSundayNote();

            try {
                await createGoogleFormFor(nextId);

                setMessage(
                    `${title} created and its Google Form is ready. Use “Copy Form Link”.`
                );
            } catch (formErr) {
                setMessage(`${title} monthly form created.`);
                setError(
                    `Google Form could not be created: ${formErr.message} Use “Create Google Form” to retry.`
                );
            }

            await loadSundaySlots(nextId);
        } catch (err) {
            setError(err.message);
        } finally {
            setCreatingMonth(false);
        }
    }

    // ---------------- CLOSE / REOPEN ----------------

    async function toggleForm(action) {
        if (!selectedId) return;

        const text =
            action === "close"
                ? "Close this monthly form? Scheduling tools will open after this."
                : "Reopen this monthly form? Scheduling tools will be hidden until it is closed again.";

        if (!window.confirm(text)) return;

        setWorking(true);
        setError("");
        setMessage("");

        let result;

        try {
            result = await request(
                `/monthly-forms/${selectedId}/${action}`,
                { method: "PATCH" }
            );
        } catch (err) {
            setError(err.message);
            setWorking(false);
            return;
        }

        // Make the Google Form match the database state.
        let googleError = "";

        try {
            await request(`/google-forms/${selectedId}/sync-status`, {
                method: "POST",
            });
        } catch (err) {
            googleError = err.message;
        }

        try {
            await loadForms();
        } catch {
            // loadForms reports its own errors.
        }

        setMessage(
            result.message ||
            (action === "close"
                ? "Monthly form closed. You can now freeze dates."
                : "Monthly form reopened.")
        );

        if (googleError) {
            setError(
                `The monthly form was updated, but the Google Form could not be ${
                    action === "close" ? "closed" : "reopened"
                }: ${googleError}`
            );
        }

        setWorking(false);
    }
    // ---------------- FROZEN DATES ----------------

    async function addFrozenDate() {
        if (!selectedId || !selectedForm) {
            setError("Select a monthly form first.");
            return;
        }

        if (!dateInput) {
            setError("Choose a date to freeze first.");
            return;
        }

        const date = new Date(`${dateInput}T12:00:00`);

        if (
            date.getMonth() + 1 !== Number(selectedForm.month) ||
            date.getFullYear() !== Number(selectedForm.year)
        ) {
            setError("Choose a date within the selected monthly form.");
            return;
        }

        // Same date is allowed with a different time
        if (isSlotFrozen(frozenDates, dateInput, freezeTimeInput)) {
            setError("That date and time has already been frozen.");
            return;
        }

        if (freezeNoteInput.length > 100) {
            setError("Frozen date note cannot exceed 100 characters.");
            return;
        }

        setWorking(true);
        setError("");
        setMessage("");

        try {
            await request("/frozen-dates/", {
                method: "POST",
                body: JSON.stringify({
                    monthly_form_id: Number(selectedId),
                    frozen_date: dateInput,
                    frozen_time: freezeTimeInput,
                    reason: freezeNoteInput.trim() || null,
                }),
            });

            await loadFrozenDates(selectedId);

            setDateInput("");
            setFreezeNoteInput("");
            setMessage("Frozen date saved successfully.");
        } catch (err) {
            console.error("Add frozen date failed:", err);
            setError(err.message);
        } finally {
            setWorking(false);
        }
    }

    async function removeFrozenDate(id) {

        setError("");
        setMessage("");

                method: "DELETE",

                cancelEditingFrozenNote();

            setMessage(result.message || "Frozen date removed successfully.");
        } catch (err) {
            setError(err.message);
        } finally {
            setWorking(false);
        }
    }

    async function runAction(action) {
        if (!scheduleUnlocked) {
            setError("Close the form and finish freezing dates first.");
            return;
        }

        const labels = {
            generate: "Generate schedule positions",
            auto: "Run automatic assignments",
        };

        if (!window.confirm(`${labels[action]} for this monthly form?`)) {
            return;

        setWorking(true);
        setError("");
        setMessage("");

        try {
            const path =
                action === "generate"
                    ? `/assignments/generate/${selectedId}`
                    : `/assignments/auto-assign/${selectedId}`;

            const result = await request(path, { method: "POST" });

            if (action === "auto") {
                setMessage(
                    `${result.message} Eligible lectors: ${result.eligible_lectors}. Assigned: ${result.assignments_created}. Left unfilled: ${result.positions_left_unfilled}. Frozen positions skipped: ${result.frozen_positions_skipped}.`
                );
            } else {
                setMessage(
                    `${result.message} Created: ${result.positions_created}. Already existed: ${result.existing_positions_skipped}.`
                );
            }

            await loadAssignments(selectedId);
        } catch (err) {
            setError(err.message);
        } finally {
            setWorking(false);
        }
    }

    async function deleteAllAssignments() {
        if (!scheduleUnlocked || assignments.length === 0) return;

                `Delete ALL ${assignments.length} assignment positions for this month, including manual assignments, frozen dates and notes, and Sunday Mass notes? You can freeze and generate again afterwards.`

            setWorking(false);

    async function saveManualAssignment(
        assignmentId,
        allowOverride = false
    ) {
        if (!manualLectorId) {
            setError("Please select a lector.");
            return;
        }

        setWorking(true);
        setError("");
        setMessage("");

        try {
            const result = await request(
                `/assignments/${assignmentId}/manual`,
                {
                    method: "PATCH",
                    body: JSON.stringify({
                        lector_id: Number(manualLectorId),
                        override_availability: allowOverride,
                    }),
                }
            );

            setMessage(
                result.message || "Manual assignment saved successfully."
            );

            cancelEditing();
            await loadAssignments(selectedId);
        } catch (err) {
            let detail = err.detail;

            if (!detail && err.message) {
                try {
                    detail = JSON.parse(err.message);
                } catch {
                    detail = null;
                }
            }

            if (
                !allowOverride &&
                detail?.requires_confirmation &&
                Array.isArray(detail.warnings)
            ) {
                const confirmed = window.confirm(
                    "Availability warnings:\n\n" +
                    detail.warnings.map((warning) => `• ${warning}`).join("\n") +
                    "\n\nDo you want to assign this lector anyway?"
                );

                if (confirmed) {
                    setWorking(false);
                    return saveManualAssignment(assignmentId, true);
                }

                setError("Assignment cancelled.");
            } else {
                setError(err.message || "Could not save assignment.");
            }
        } finally {
            setWorking(false);
        }
    }

    // ---------------- DOWNLOADS ----------------

    function downloadRoster(format) {
        if (!selectedForm || assignments.length === 0) {
            setError("Load assignments before downloading the Lectors List.");
            return;
        }

        const html = buildRosterDocument(
            selectedForm,
            assignments,
            sundaySlots,
            frozenDates
        );
        const monthLabel =
            selectedForm.title ||
            `${selectedForm.month || "month"}-${selectedForm.year || ""}`;

        const safeName = String(monthLabel)
            .trim()
            .replace(/[^a-z0-9-_]+/gi, "-");

        if (format === "word") {
            const blob = new Blob(["\ufeff", html], {
                type: "application/msword;charset=utf-8",
            });

            const url = URL.createObjectURL(blob);
            const link = document.createElement("a");

            link.href = url;
            link.download = `Lectors-List-${safeName}.doc`;

            document.body.appendChild(link);
            link.click();
            link.remove();

            URL.revokeObjectURL(url);
            return;
        }

        const printWindow = window.open("", "_blank");

        if (!printWindow) {
            setError("Allow pop-ups for this site to create the PDF.");
            return;
        }

        printWindow.document.open();
        printWindow.document.write(html);
        printWindow.document.close();

        printWindow.onload = () => {
            printWindow.focus();
            printWindow.print();
        };
    }

                                className="secondary-button"

                            >
                                <Pencil size={16} />
                                Edit Frozen Dates
                            </button>

                            <button
                                type="button"
                                className="primary-button"
                                onClick={() => persistFreezeDone(true)}
                                disabled={working}
                            >
                                <CheckCircle2 size={16} />
                                Done Freezing
                            </button>
                        )}
                    </div>

                    <p className="field-hint">
                        {freezeDone
                            ? "Frozen dates are locked in. Continue to Step 2."
                            : "Only the chosen time is frozen; other Masses that day can still be auto-assigned. Click “Done Freezing” when finished."}
                    </p>
                </section>
            )}

            {/* STEP 2: GENERATE / ASSIGN / DELETE */}
            {showScheduling && (
                <section
                    className="content-card"
                    style={{ opacity: freezeDone ? 1 : 0.55 }}
                >
                    <div className="section-heading">
                        <div>
                            <h2>Step 2 · Generate &amp; Assign</h2>
                            <p>
                                Create positions, auto-assign eligible
                                lectors, or delete everything and start again.
                            </p>
                        </div>
                        <WandSparkles size={25} className="soft-icon" />
                    </div>

                    <div className="schedule-actions">
                        <button
                            className="secondary-button"
                            disabled={working || !scheduleUnlocked}
                            onClick={() => runAction("generate")}
                        >
                            <CalendarDays size={17} />
                            Generate Positions
                        </button>

                        <button
                            className="primary-button"
                            disabled={working || !scheduleUnlocked}
                            onClick={() => runAction("auto")}
                        >
                            <WandSparkles size={17} />
                            {working ? "Working..." : "Auto-Assign Lectors"}
                        </button>

                        <button
                            className="secondary-button"
                            disabled={
                                working ||
                                !scheduleUnlocked ||
                                assignments.length === 0
                            }
                            onClick={deleteAllAssignments}
                        >
                            <Trash2 size={17} />
                            Delete All Assignments
                        </button>
                    </div>

                    <p className="field-hint">
                        {freezeDone
                            ? "Positions and automatic lector selection are handled by your backend."
                            : "Click “Done Freezing” in Step 1 to unlock this step."}
                    </p>
                </section>
            )}

            {/* STEP 3: SUMMARY + CALENDAR */}
            {showScheduling && (
                <>
                    <div className="schedule-summary-grid">
                        <div className="summary-card">
                            <CalendarDays size={20} />
                            <span>Total Positions</span>
                            <strong>{assignments.length}</strong>
                        </div>

                        <div className="summary-card">
                            <CheckCircle2 size={20} />
                            <span>Filled Positions</span>
                            <strong>{filled}</strong>
                        </div>

                        <div className="summary-card">
                            <CircleAlert size={20} />
                            <span>Unfilled Positions</span>
                            <strong>{unfilled}</strong>
                        </div>
                    </div>

                    <section className="content-card">
                        <div className="section-heading">
                            <div>
                                <h2>Assignment Progress</h2>
                                <p>
                                    Live progress based on saved backend
                                    assignments.
                                </p>
                            </div>
                            <strong>{progress}%</strong>
                        </div>

                        <div className="progress-track">
                            <div
                                className="progress-fill"
                                style={{ width: `${progress}%` }}
                            />
                        </div>
                    </section>

                    {/* MONTHLY CALENDAR */}
                    <section className="content-card">
                        <div className="section-heading">
                            <div>
                                <h2>
                                    Step 3 · Monthly Schedule
                                    {selectedForm?.title
                                        ? ` — ${selectedForm.title}`
                                        : ""}
                                </h2>
                                <p>
                                    Click a date to view its assignments.
                                    Click Sunday 7:30 AM or 9:45 AM to edit

                                </p>
                            </div>
                            <CalendarDays size={25} className="soft-icon" />

                        {assignments.length === 0 &&
                            sundaySlots.length === 0 &&
                            frozenDates.length === 0 ? (
                            <p className="empty-state">
                                No assignment positions or Sunday Mass slots

                                that the Sunday slots API is enabled.
                            </p>
                        ) : (
                            <div className="cal-grid-wrap">
                                <div className="cal-grid">

                                            {day}

                                            frozenForDay.map(getFrozenTime)

                                            (item) =>
                                                !frozenTimes.has(

                                                        getMassTime(item)
                                                    )
                                                )
                                        );

                                        const fullyFrozen =
                                            frozenForDay.length > 0 &&
                                            openItems.length === 0;

                                        const total = openItems.length;

                                        const done = openItems.filter(
                                            (item) =>
                                                item.lector_id != null ||
                                                Boolean(item.special_text)

                                        const byTime = [];

                                        openItems.forEach((item) => {
                                            const time = formatMassTime(
                                                getMassTime(item)
                                            );

                                            let group = byTime.find(
                                                (entry) => entry.time === time
                                            );

                                            if (!group) {
                                                group = { time, items: [] };
                                                byTime.push(group);
                                            }

                                            group.items.push(item);
                                        });

                                        const sundaySlotsForDay =
                                            cell.sundaySlots || [];

                                        return (
                                            <div
                                                key={cell.dateStr}
                                                className={`cal-cell ${selectedDay === cell.dateStr
                                                    ? "cal-selected"
                                                    : ""
                                                    } ${fullyFrozen ? "cal-frozen" : ""}`}
                                                onClick={() => {
                                                    setSelectedDay(cell.dateStr);
                                                    cancelEditing();
                                                    cancelEditingFrozenNote();
                                                    cancelEditingSundayNote();
                                                }}
                                            >
                                                <div className="cal-date-row">
                                                    <span className="cal-date">
                                                        {cell.dayNumber}
                                                    </span>

                                                    <span className="cal-badges">
                                                        {total > 0 && (
                                                            <span
                                                                className={`cal-badge ${done === total
                                                                    ? "cal-badge-ok"
                                                                    : "cal-badge-warn"
                                                                    }`}
                                                            >
                                                                {done}/{total}
                                                            </span>
                                                        )}
                                                    </span>
                                                </div>

                                                <div className="cal-lines">
                                                    {/* One line per frozen time */}
                                                    {frozenForDay.map((f) => (
                                                        <div
                                                            className="cal-time-group"
                                                            key={`frozen-${f.id}`}
                                                        >
                                                            <span className="cal-line cal-line-frozen">
                                                                {getFrozenTime(f)} —{" "}
                                                                {f.reason || "Frozen"}
                                                            </span>
                                                        </div>
                                                    ))}

                                                    {/* Sunday special Mass note buttons */}
                                                    {sundaySlotsForDay.length > 0 &&
                                                        SUNDAY_SPECIAL_TIMES.map(
                                                            (time) => {
                                                                const slot =
                                                                    sundaySlotsForDay.find(
                                                                        (item) =>
                                                                            isSundaySlot(
                                                                                item,
                                                                                time
                                                                            )
                                                                    );

                                                                return (
                                                                    <button
                                                                        type="button"
                                                                        key={`${cell.dateStr}-${time}`}
                                                                        className="cal-line sunday-slot-note-button"

                                                                            setSelectedDay(
                                                                                cell.dateStr

                                                                            cancelEditing();
                                                                            cancelEditingFrozenNote();
                                                                            startEditingSundayNote(
                                                                                slot
                                                                            );

                                                                        title={`Add or edit ${time} note`}

                                                                        <b>{time}</b>
                                                                        <span>
                                                                            {slot.note
                                                                                ? slot.note
                                                                                : "＋ Add note"}
                                                                        </span>
                                                                    </button>
                                                                );
                                                            }
                                                        )}

                                                    {/* Open (non-frozen) times, e.g. 6:00 PM */}
                                                    {byTime.map((group) => (
                                                        <div
                                                            className="cal-time-group"
                                                            key={group.time}
                                                        >
                                                            <span className="cal-time">
                                                                {group.time}
                                                            </span>

                                                            {group.items.map((item) => (
                                                                <span
                                                                    key={item.id}
                                                                    className={`cal-line ${item.special_text

                                                                    <b>
                                                                        {getReadingLabel(item) || "—"}

                                                                    {item.lector_id != null ||
                                                                        item.special_text
                                                                        ? getAssignmentName(item)
                                                                        : "—"}
                                                                </span>
                                                            ))}
                                                        </div>
                                                    ))}
                                                </div>
                                            </div>
                                        );
                                    })}
                                </div>
                            </div>
                        )}
                    </section>

                    {/* DAY DETAIL / EDIT PANEL */}
                    {selectedDay && (
                        <section className="content-card">
                            <div className="section-heading">
                                <div>
                                    <h2>
                                        {new Date(
                                            `${selectedDay}T12:00:00`
                                        ).toLocaleDateString("en-GB", {
                                            weekday: "long",
                                            day: "numeric",
                                            month: "long",
                                            year: "numeric",
                                        })}
                                    </h2>

                                    <p>
                                        View the Sunday Mass notes, Frozen Date
                                        Notes, and normal assignment positions
                                        separately.
                                    </p>
                                </div>

                                <button
                                    type="button"
                                    className="secondary-button"
                                    onClick={() => {
                                        setSelectedDay(null);
                                        cancelEditing();
                                        cancelEditingFrozenNote();
                                        cancelEditingSundayNote();
                                    }}
                                >
                                    <X size={15} />
                                    Close
                                </button>
                            </div>

                            {/* SUNDAY 7:30 AM AND 9:45 AM NOTE EDITORS */}
                            {selectedDaySundaySlots.length > 0 && (
                                <div className="calendar-mass-list">
                                    {SUNDAY_SPECIAL_TIMES.map((time) => {
                                        const slot =
                                            selectedDaySundaySlots.find(
                                                (item) =>
                                                    isSundaySlot(item, time)
                                            );

                                        if (!slot) return null;

                                        const isEditingThisSlot =
                                            editingSundayNote &&
                                            selectedSundaySlot &&
                                            selectedSundaySlot.mass_slot_id ===
                                            slot.mass_slot_id &&
                                            getSundaySlotDate(
                                                selectedSundaySlot
                                            ) === getSundaySlotDate(slot);

                                        return (
                                            <div
                                                className="calendar-assignment"
                                                key={`${selectedDay}-${time}`}
                                            >
                                                <div className="calendar-assignment-main">
                                                    <div className="calendar-assignment-time">
                                                        <strong>{time}</strong>
                                                        <span>Sunday Special Mass</span>
                                                    </div>

                                                    <div className="calendar-assignment-info">
                                                        <strong>Mass Note</strong>
                                                        <span>
                                                            {slot.note ||
                                                                "No note added."}
                                                        </span>
                                                    </div>

                                                    <span className="status-pill status-active">
                                                        No Positions
                                                    </span>
                                                </div>

                                                {!isEditingThisSlot && (
                                                    <div className="calendar-assignment-actions">
                                                        <button
                                                            type="button"
                                                            className="secondary-button"
                                                            onClick={() =>
                                                                startEditingSundayNote(
                                                                    slot
                                                                )
                                                            }
                                                            disabled={working}
                                                        >
                                                            <Pencil size={15} />
                                                            {slot.note
                                                                ? "Edit Note"
                                                                : "Add Note"}
                                                        </button>
                                                    </div>
                                                )}

                                                {isEditingThisSlot && (
                                                    <div className="manual-editor">
                                                        <h4>
                                                            {time} Mass Note
                                                        </h4>

                                                        <label
                                                            htmlFor={`sunday-note-${time.replace(
                                                                /[^0-9]/g,
                                                                ""
                                                            )}`}
                                                        >
                                                            Enter note
                                                        </label>

                                                        <textarea
                                                            id={`sunday-note-${time.replace(
                                                                /[^0-9]/g,
                                                                ""
                                                            )}`}
                                                            className="form-select"
                                                            rows={3}
                                                            maxLength={1000}
                                                            value={sundayNoteDraft}
                                                            onChange={(e) =>
                                                                setSundayNoteDraft(
                                                                    e.target.value
                                                                )
                                                            }
                                                            placeholder={`Enter a note for the Sunday ${time} Mass`}
                                                            disabled={working}
                                                        />

                                                        <p className="field-hint">
                                                            This note is saved
                                                            separately for this
                                                            Sunday date and Mass
                                                            time. It does not
                                                            create assignment
                                                            positions.
                                                        </p>

                                                        <div className="schedule-actions">
                                                            <button
                                                                type="button"
                                                                className="primary-button"
                                                                onClick={saveSundayNote}
                                                                disabled={working}
                                                            >
                                                                {working
                                                                    ? "Saving..."
                                                                    : "Save Note"}
                                                            </button>

                                                            <button
                                                                type="button"
                                                                className="secondary-button"
                                                                onClick={cancelEditingSundayNote}
                                                                disabled={working}
                                                            >
                                                                Cancel
                                                            </button>
                                                        </div>
                                                    </div>
                                                )}
                                            </div>
                                        );
                                    })}
                                </div>
                            )}

                            {/* FROZEN DATE NOTE EDITORS - ONE PER FROZEN TIME */}
                            {selectedFrozenDates.map((fd) => (
                                <div className="empty-state" key={fd.id}>
                                    {editingFrozenId === fd.id ? (
                                        <div style={{ width: "100%" }}>
                                            <label htmlFor={`frozen-time-${fd.id}`}>
                                                Time
                                            </label>

                                            <select
                                                id={`frozen-time-${fd.id}`}
                                                className="form-select"
                                                value={frozenTimeDraft}
                                                onChange={(e) =>
                                                    setFrozenTimeDraft(
                                                        e.target.value
                                                    )
                                                }
                                                disabled={working}
                                            >
                                                {ROSTER_SLOTS.map((time) => (
                                                    <option
                                                        key={time}
                                                        value={time}
                                                    >
                                                        {time}
                                                    </option>
                                                ))}
                                            </select>

                                            <label htmlFor={`frozen-note-${fd.id}`}>
                                                Frozen Date Note
                                            </label>

                                            <input
                                                id={`frozen-note-${fd.id}`}
                                                type="text"
                                                autoFocus
                                                className="form-select"
                                                maxLength={100}
                                                placeholder="Enter a note, e.g. St. Anne's Community"
                                                value={frozenNoteDraft}
                                                onChange={(e) =>
                                                    setFrozenNoteDraft(
                                                        e.target.value
                                                    )
                                                }
                                                disabled={working}
                                                onKeyDown={(e) => {
                                                    if (e.key === "Enter") {
                                                        e.preventDefault();
                                                        saveFrozenNote();
                                                    }

                                                    if (e.key === "Escape") {
                                                        cancelEditingFrozenNote();
                                                    }
                                                }}
                                            />

                                            <p className="field-hint">
                                                Maximum 100 characters. The
                                                note appears in the selected
                                                time column of the downloaded
                                                roster.
                                            </p>

                                            <div className="schedule-actions">
                                                <button
                                                    type="button"
                                                    className="primary-button"
                                                    onClick={saveFrozenNote}
                                                    disabled={working}
                                                >
                                                    {working
                                                        ? "Saving..."
                                                        : "Save Note"}
                                                </button>

                                                <button
                                                    type="button"
                                                    className="secondary-button"
                                                    onClick={cancelEditingFrozenNote}
                                                    disabled={working}
                                                >
                                                    Cancel
                                                </button>
                                            </div>
                                        </div>
                                    ) : (
                                        <div style={{ width: "100%" }}>
                                            <p>
                                                <strong>
                                                    Frozen Date Note (
                                                    {getFrozenTime(fd)}
                                                    ):
                                                </strong>{" "}
                                                {fd.reason || "No note added."}
                                            </p>

                                            <div className="schedule-actions">
                                                <button
                                                    type="button"
                                                    className="secondary-button"
                                                    onClick={() =>
                                                        startEditingFrozenNote(fd)
                                                    }
                                                    disabled={working}
                                                >
                                                    <Pencil size={15} />
                                                    Edit Frozen Date Note
                                                </button>
                                            </div>
                                        </div>
                                    )}
                                </div>
                            ))}

                            {/* NORMAL ASSIGNMENT POSITIONS */}
                            {selectedDayItems.length === 0 ? (
                                <p className="empty-state">
                                    No normal assignment positions on this date.
                                </p>
                            ) : (
                                <div className="calendar-mass-list">
                                    {selectedDayItems.map((item) => {
                                        const isEditing =
                                            editingAssignmentId === item.id;

                                        const blocked = !scheduleUnlocked;

                                        const itemTime = formatMassTime(
                                            getMassTime(item)
                                        );

                                        const itemFrozen = isSlotFrozen(
                                            frozenDates,
                                            selectedDay,
                                            itemTime
                                        );

                                        return (
                                            <div
                                                className="calendar-assignment"
                                                key={item.id}
                                            >
                                                <div className="calendar-assignment-main">
                                                    <div className="calendar-assignment-time">
                                                        <strong>
                                                            {itemTime}
                                                        </strong>

                                                        <span>
                                                            {item.mass_type ||
                                                                item.mass_slot?.mass_type ||
                                                                "Mass"}
                                                        </span>
                                                    </div>

                                                    <div className="calendar-assignment-info">
                                                        <strong>
                                                            {item.responsibility}
                                                        </strong>

                                                        <span>
                                                            Position{" "}
                                                            {item.position_number}
                                                        </span>

                                                        <span
                                                            className={
                                                                item.lector_id != null ||
                                                                    item.special_text
                                                                    ? "assignment-lector"
                                                                    : "assignment-unfilled"
                                                            }
                                                        >
                                                            <UsersRound size={15} />
                                                            {getAssignmentName(item)}
                                                        </span>
                                                    </div>

                                                    <span
                                                        className={`status-pill ${item.lector_id != null ||
                                                            item.special_text
                                                            ? "status-active"
                                                            : "status-pending"
                                                            }`}
                                                    >
                                                        {item.special_text
                                                            ? "Custom Text"
                                                            : item.lector_id != null
                                                                ? "Assigned"
                                                                : "Unassigned"}
                                                    </span>
                                                </div>

                                                <div className="calendar-assignment-actions">
                                                    {itemFrozen && (
                                                        <span className="field-hint">
                                                            Frozen time
                                                        </span>
                                                    )}

                                                    {item.is_manual && (
                                                        <span className="field-hint">
                                                            Manual
                                                        </span>
                                                    )}

                                                    {item.is_protected && (
                                                        <span className="field-hint">
                                                            Protected
                                                        </span>
                                                    )}

                                                    <button
                                                        type="button"
                                                        className="secondary-button"
                                                        disabled={working || blocked}
                                                        onClick={() =>
                                                            startEditing(item)
                                                        }
                                                    >
                                                        <Pencil size={15} />
                                                        Edit
                                                    </button>

                                                    <button
                                                        type="button"
                                                        className="secondary-button"
                                                        disabled={working || blocked}
                                                        onClick={() =>
                                                            clearManualAssignment(
                                                                item.id
                                                            )
                                                        }
                                                    >
                                                        <Trash2 size={15} />
                                                        Clear Position
                                                    </button>
                                                </div>

                                                {isEditing && (
                                                    <div className="manual-editor">
                                                        <h4>
                                                            Edit Assignment Position
                                                        </h4>

                                                        <div className="schedule-actions">
                                                            <button
                                                                type="button"
                                                                className={
                                                                    assignmentMode ===
                                                                        "lector"
                                                                        ? "primary-button"
                                                                        : "secondary-button"
                                                                }
                                                                onClick={() =>
                                                                    setAssignmentMode(
                                                                        "lector"
                                                                    )
                                                                }
                                                                disabled={working}
                                                            >
                                                                <UsersRound size={16} />
                                                                Assign Lector
                                                            </button>

                                                            <button
                                                                type="button"
                                                                className={
                                                                    assignmentMode ===
                                                                        "text"
                                                                        ? "primary-button"
                                                                        : "secondary-button"
                                                                }
                                                                onClick={() =>
                                                                    setAssignmentMode(
                                                                        "text"
                                                                    )
                                                                }
                                                                disabled={working}
                                                            >
                                                                <Pencil size={16} />
                                                                Custom Text
                                                            </button>
                                                        </div>

                                                        {assignmentMode === "lector" ? (
                                                            <>
                                                                <label>
                                                                    Select an active lector
                                                                </label>

                                                                <select
                                                                    className="form-select"
                                                                    value={manualLectorId}
                                                                    onChange={(e) =>
                                                                        setManualLectorId(
                                                                            e.target.value
                                                                        )
                                                                    }
                                                                >
                                                                    <option value="">
                                                                        Choose a lector
                                                                    </option>

                                                                    {activeLectors.map(
                                                                        (lector) => (
                                                                            <option
                                                                                key={lector.id}
                                                                                value={lector.id}
                                                                            >
                                                                                {getLectorName(
                                                                                    lector,
                                                                                    lector.id
                                                                                )}
                                                                            </option>
                                                                        )
                                                                    )}
                                                                </select>

                                                                <p className="field-hint">
                                                                    Hard scheduling
                                                                    conflicts, such
                                                                    as consecutive
                                                                    serving days,
                                                                    are still checked
                                                                    by the backend.
                                                                </p>

                                                                <div className="schedule-actions">
                                                                    <button
                                                                        type="button"
                                                                        className="primary-button"
                                                                        disabled={
                                                                            working ||
                                                                            !manualLectorId
                                                                        }
                                                                        onClick={() =>
                                                                            saveManualAssignment(
                                                                                item.id

                                                                        }
                                                                    >
                                                                        Save Assignment
                                                                    </button>
                                                                </div>
                                                            </>
                                                        ) : (
                                                            <>
                                                                <label>
                                                                    Enter Custom Text
                                                                </label>

                                                                <input
                                                                    type="text"
                                                                    className="form-select"
                                                                    value={specialText}
                                                                    onChange={(e) =>
                                                                        setSpecialText(
                                                                            e.target.value
                                                                        )
                                                                    }
                                                                    maxLength={100}
                                                                    placeholder="e.g. Zone 1, Sunday School, Youth Group"
                                                                    disabled={working}
                                                                />

                                                                <p className="field-hint">
                                                                    Custom text
                                                                    replaces the
                                                                    lector name in
                                                                    this existing
                                                                    position.
                                                                    Maximum 100
                                                                    characters.
                                                                </p>

                                                                <div className="schedule-actions">
                                                                    <button
                                                                        type="button"
                                                                        className="primary-button"
                                                                        disabled={
                                                                            working ||
                                                                            !specialText.trim()
                                                                        }
                                                                        onClick={() =>
                                                                            saveSpecialText(
                                                                                item.id
                                                                            )
                                                                        }
                                                                    >
                                                                        Save Custom Text
                                                                    </button>

                                                                    <button
                                                                        type="button"
                                                                        className="secondary-button"
                                                                        disabled={
                                                                            working ||
                                                                            !item.special_text
                                                                        }
                                                                        onClick={() =>
                                                                            saveSpecialText(
                                                                                item.id,
                                                                                ""
                                                                            )
                                                                        }
                                                                    >
                                                                        Clear Custom Text
                                                                    </button>
                                                                </div>
                                                            </>
                                                        )}

                                                        <div className="schedule-actions">
                                                            <button
                                                                type="button"
                                                                className="secondary-button"
                                                                disabled={working}
                                                                onClick={cancelEditing}
                                                            >
                                                                Cancel
                                                            </button>
                                                        </div>
                                                    </div>
                                                )}
                                            </div>
                                        );
                                    })}
                                </div>
                            )}
                        </section>
                    )}

                    {/* DOWNLOAD */}
                    <section className="content-card">
                        <div className="section-heading">
                            <div>
                                <h2>Download Lectors List</h2>
                                <p>
                                    Export the assignments in your printable
                                    church roster format.
                                </p>
                            </div>
                            <Download size={25} className="soft-icon" />
                        </div>

                        <div className="schedule-actions roster-download-actions">
                            <button
                                type="button"
                                className="secondary-button"
                                onClick={() => downloadRoster("pdf")}
                                disabled={assignments.length === 0}
                            >
                                <FileText size={17} />
                                Download PDF
                            </button>

                            <button
                                type="button"
                                className="primary-button"
                                onClick={() => downloadRoster("word")}
                                disabled={assignments.length === 0}
                            >
                                <Download size={17} />
                                Download Word
                            </button>
                        </div>

                        <p className="field-hint">
                            The printable Date, Day, Readings and Mass time
                            table is generated when you download the roster.
                        </p>
                    </section>
                </>
            )}
        </div>
    );
}
