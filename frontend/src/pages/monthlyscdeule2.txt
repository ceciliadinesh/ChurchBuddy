    "7:00 AM",
    "7:30 AM",

    "Ang.",
    "Rdng.1",

// --------------------------------------------------
// GENERAL HELPERS

function getList(data) {
    return Array.isArray(data) ? data : data?.items || [];
}

function getLectorName(lector, lectorId) {
    if (!lector) {
        return lectorId != null
            ? `Lector #${lectorId}`
            : "Unassigned";
    }

    const firstName =
        lector.first_name || lector.firstname || "";

    const lastName =
        lector.last_name || lector.lastname || "";

    const combined = `${firstName} ${lastName}`.trim();

    return (
        lector.full_name ||
        lector.name ||
        combined ||
        lector.email ||
        `Lector #${lectorId}`
    );
}

function getAssignmentName(item) {
    return (
        item.lector_name ||
        getLectorName(item.lector, item.lector_id)
    );
}

function getMassTime(item) {
    return (
        item.mass_slot?.mass_time ||
        item.mass_slot?.time ||
        item.mass_time ||
        item.slot_time ||
        ""
    );
}

function formatMassTime(value) {
    if (!value) return "Time unavailable";

    const parts = String(value).split(":");
    const hour = Number(parts[0]);
    const minute = parts[1] || "00";

    if (Number.isNaN(hour)) return value;

    const period = hour >= 12 ? "PM" : "AM";
    const displayHour = hour % 12 || 12;

    return `${displayHour}:${minute} ${period}`;
}

function getReadingLabel(item) {
    const value = String(
        item.reading ||
        item.reading_type ||
        item.responsibility ||
        item.position_name ||
        item.position ||
        ""
    ).trim();

    if (!value) return "";

    const normalized = value.toLowerCase().replace(/\s+/g, "");

    const aliases = {
        ang: "Ang.",
        angelus: "Ang.",
        "rdng.1": "Rdng.1",
        reading1: "Rdng.1",
        "rdng.2": "Rdng.2",
        reading2: "Rdng.2",
        ps: "P.S.",
        "p.s.": "P.S.",
        pof: "POF",
    };

    return aliases[normalized] || value;
}

function formatRosterDate(dateValue) {
    if (!dateValue) {
        return { date: "", day: "" };
    }

    const date = new Date(
        `${String(dateValue).slice(0, 10)}T12:00:00`
    );

    if (Number.isNaN(date.getTime())) {
        return { date: String(dateValue), day: "" };
    }

    return {
        date: date
            .toLocaleDateString("en-GB", {
                day: "2-digit",
                month: "short",
            })
            .toUpperCase(),

        day: date
            .toLocaleDateString("en-US", {
                weekday: "short",
            })
            .toUpperCase(),
    };
}

function escapeHtml(value) {
    return String(value ?? "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}

function getDateOnly(value) {
    return String(value || "").slice(0, 10);
}

// --------------------------------------------------
// CALENDAR GROUPING
// --------------------------------------------------

function groupCalendarAssignments(assignments) {
    const groups = {};

    assignments.forEach((item) => {
        const date = getDateOnly(
            item.assignment_date || item.date
        );

        if (!date) return;

        if (!groups[date]) {
            groups[date] = [];
        }

        groups[date].push(item);
    });

    return Object.entries(groups)
        .sort(([a], [b]) => a.localeCompare(b))
        .map(([date, items]) => ({
            date,
            ...formatRosterDate(date),

            day: new Date(
                `${date}T12:00:00`
            ).toLocaleDateString("en-US", {
                weekday: "long",
            }),

            slots: [...items].sort((a, b) => {
                const timeA = getMassTime(a);
                const timeB = getMassTime(b);

                if (timeA !== timeB) {
                    return timeA.localeCompare(timeB);
                }

                return (
                    (a.position_number || 0) -
                    (b.position_number || 0)
                );
            }),
        }));
}

function getRosterSlotIndex(item) {
    const raw = String(getMassTime(item))
        .toLowerCase()
        .replace(/\s+/g, "");

    const compact = raw.replace(/[^a-z0-9]/g, "");

    if (
        compact.includes("700am") ||
        compact.includes("7am") ||
        compact.includes("0700")
    ) {
        return 0;
    }

    if (
        compact.includes("730am") ||
        compact.includes("730") ||
        compact.includes("0730")
    ) {
        return 1;
    }

    if (
        compact.includes("945am") ||
        compact.includes("945") ||
        compact.includes("0945")
    ) {
        return 2;
    }

    if (
        compact.includes("600pm") ||
        compact.includes("6pm") ||
        compact.includes("1800")
    ) {
        return 3;
    }

    return ROSTER_SLOTS.findIndex(
        (slot) =>
            raw === slot.toLowerCase().replace(/\s+/g, "")
    );
}

function buildRosterGroups(assignments) {
    const byDate = new Map();

    assignments.forEach((item) => {
        const date = getDateOnly(
            item.assignment_date || item.date
        );

        if (!date) return;

        if (!byDate.has(date)) {
            byDate.set(date, []);
        }

        byDate.get(date).push(item);
    });

    return [...byDate.entries()]
        .sort(([a], [b]) => a.localeCompare(b))
        .map(([dateValue, items]) => {
            const slotItems = [[], [], [], []];
            const readingLabels = new Set();

            items.forEach((item) => {
                const label = getReadingLabel(item);

                if (label) {
                    readingLabels.add(label);
                }

                const slot = getRosterSlotIndex(item);

                if (slot >= 0) {
                    slotItems[slot].push(item);
                }
            });

            const labels = [...readingLabels].sort((a, b) => {
                const ai = READING_ORDER.indexOf(a);
                const bi = READING_ORDER.indexOf(b);

                if (ai === -1 && bi === -1) {
                    return a.localeCompare(b);
                }

                if (ai === -1) return 1;
                if (bi === -1) return -1;

                return ai - bi;
            });

            const rowCount = Math.max(
                1,
                labels.length,
                ...slotItems.map((slot) => slot.length)
            );

            const rows = Array.from(
                { length: rowCount },
                (_, index) => ({
                    reading: labels[index] || "",
                    slots: slotItems.map((slot) =>
                        slot[index]
                            ? getAssignmentName(slot[index])
                            : ""
                    ),
                })
            );

            const unknownSlotItems = items.filter(
                (item) => getRosterSlotIndex(item) < 0
            );

            unknownSlotItems.forEach((item, index) => {
                const rowIndex = index % rows.length;

                const extra = [
                    getReadingLabel(item),
                    getAssignmentName(item),
                ]
                    .filter(Boolean)
                    .join(" — ");

                rows[rowIndex].slots[0] = [
                    rows[rowIndex].slots[0],
                    extra,
                ]
                    .filter(Boolean)
                    .join("; ");
            });

            return {
                dateValue,
                ...formatRosterDate(dateValue),
                rows,
            };
        });
}

function buildRosterDocument(form, assignments) {
    const groups = buildRosterGroups(assignments);

    const monthYear =
        form?.title ||
        (form?.month && form?.year
            ? `${new Date(
                Number(form.year),
                Number(form.month) - 1,
                1
            ).toLocaleDateString("en-US", {
                month: "long",
            })} ${form.year}`
            : "MONTH YEAR");

    const bodyRows = groups
        .map((group) =>
            group.rows
                .map(
                    (row, index) => `
      <tr class="${index === 0 ? "date-start" : ""}">
        ${
            index === 0
                ? `
              <td class="date" rowspan="${group.rows.length}">
                ${escapeHtml(group.date)}
              </td>
              <td class="day" rowspan="${group.rows.length}">
                ${escapeHtml(group.day)}
              </td>
            `
                : ""
        }

        <td class="reading">
          ${escapeHtml(row.reading)}
        </td>

        ${row.slots
            .map((name) => {
                const special =
                    /community|school|liturgy team|youth|sodality/i.test(
                        name
                    );

                return `
            <td class="${special ? "special" : "person"}">
              ${escapeHtml(name)}
            </td>
          `;
            })
            .join("")}
      </tr>
    `
                )
                .join("")
        )
        .join("");

    return `<!doctype html>
<html>
<head>
<meta charset="utf-8">
<title>Lectors List - ${escapeHtml(monthYear)}</title>

<style>
  @page {

    box-sizing: border-box;
  }

  body {
    margin: 0;
    padding: 18px;
    color: #111;
    background: #fff;
    font-family: Arial, Helvetica, sans-serif;
  }

  h1 {
    margin: 0 0 18px;
    text-align: center;
    font-size: 21px;
    font-weight: 700;
  }

  table {
    width: 100%;
    border-collapse: collapse;
    table-layout: fixed;
    font-size: 12px;
    background: #fff;
  }

  th, td {
    border: 1px solid #222;
    padding: 5px 6px;
    height: 27px;
    vertical-align: middle;
  }

  th {
    background: #dedede;
    text-align: center;
    font-weight: 700;
    border: 2px solid #111;
  }

  td.date, td.day {
    text-align: center;
    vertical-align: middle;
    background: #f1f1f1;
  }

  td.reading {
    text-align: left;
    width: 8%;
  }

  td.person {
    text-align: left;
    overflow-wrap: anywhere;
  }

// MAIN COMPONENT

export default function MonthlyScheduling() {
    const [forms, setForms] = useState([]);
    const [lectors, setLectors] = useState([]);
    const [selectedId, setSelectedId] = useState("");

    const [assignments, setAssignments] = useState([]);

    const [editingAssignmentId, setEditingAssignmentId] =
        useState(null);

    const [manualLectorId, setManualLectorId] = useState("");
    const [overrideAvailability, setOverrideAvailability] =
        useState(false);

    const [frozenDates, setFrozenDates] = useState([]);
    const [dateInput, setDateInput] = useState("");

    const [loading, setLoading] = useState(true);
    const [working, setWorking] = useState(false);
    const [error, setError] = useState("");
    const [message, setMessage] = useState("");

    // Add Month state
    const [newMonth, setNewMonth] = useState(
        String(new Date().getMonth() + 1)
    );

    const [newYear, setNewYear] = useState(
        String(Math.max(2026, new Date().getFullYear()))
    );

    const [creatingMonth, setCreatingMonth] = useState(false);

    // --------------------------------------------------
    // API
    // --------------------------------------------------

    function headers() {
        return {
            "Content-Type": "application/json",
            Authorization: `Bearer ${sessionStorage.getItem(
                "admin_token"
            )}`,
        };
    }

    async function request(path, options = {}) {
        const response = await fetch(`${API_URL}${path}`, {
            ...options,
            headers: {
                ...headers(),
                ...options.headers,
            },
        });

        const data = await response.json().catch(() => ({}));

        if (!response.ok) {
            const detail =
                typeof data.detail === "string"
                    ? data.detail
                    : data.detail?.message;

            throw new Error(detail || "Request failed.");
        }

        return data;
    }

    // --------------------------------------------------
    // LOAD DATA
    // --------------------------------------------------

    async function loadAssignments(id) {
        if (!id) {
            setAssignments([]);
            return;
        }

        const data = await request(`/assignments/${id}`);

        setAssignments(getList(data));
    }

    async function loadFrozenDates(id) {
        if (!id) {
            setFrozenDates([]);
            return;
        }

        const data = await request(`/frozen-dates/${id}`);

        setFrozenDates(getList(data));
    }

    async function loadForms() {
        setLoading(true);
        setError("");
        setMessage("");

        try {
            const [formData, lectorData] = await Promise.all([
                request("/monthly-forms/"),
                request("/lectors/"),
            ]);

            const formList = getList(formData);

            setForms(formList);
            setLectors(getList(lectorData));

            if (formList.length === 0) {
                setSelectedId("");
                setAssignments([]);
                setFrozenDates([]);
                return;
            }

            const existingSelection = formList.find(
                (form) => String(form.id) === String(selectedId)
            );

            const nextId = existingSelection
                ? String(existingSelection.id)
                : String(formList[0].id);

            setSelectedId(nextId);

            await Promise.all([
                loadAssignments(nextId),
                loadFrozenDates(nextId),
            ]);
        } catch (err) {
            setError(err.message);
        } finally {
            setLoading(false);
        }
    }

    useEffect(() => {
        loadForms();

        // Initial load.
        // eslint-disable-next-line react-hooks/exhaustive-deps
    }, []);

    async function selectForm(id) {
        setSelectedId(id);
        setMessage("");
        setError("");
        setDateInput("");

        setEditingAssignmentId(null);
        setManualLectorId("");

        try {
            await Promise.all([
                loadAssignments(id),
                loadFrozenDates(id),
            ]);
        } catch (err) {
            setError(err.message);
        }
    }

    // --------------------------------------------------
    // ADD MONTH
    // --------------------------------------------------

    async function createMonth() {
        const month = Number(newMonth);
        const year = Number(newYear);

        if (!Number.isInteger(month) || month < 1 || month > 12) {
            setError("Please select a valid month.");
            return;
        }

        if (!Number.isInteger(year) || year < 2026) {
            setError("Year must be 2026 or later.");
            return;
        }

        const title = new Date(year, month - 1, 1)
            .toLocaleDateString("en-US", {
                month: "long",
                year: "numeric",
            });

        setCreatingMonth(true);
        setError("");
        setMessage("");

        try {
            const result = await request("/monthly-forms/", {
                method: "POST",
                body: JSON.stringify({
                    month,
                    year,
                    title,
                    is_open: true,
                }),
            });

            const nextId = String(result.id);

            setForms((previous) =>
                [...previous, result].sort(
                    (a, b) =>
                        b.year - a.year || b.month - a.month
                )
            );

            setSelectedId(nextId);
            setAssignments([]);
            setFrozenDates([]);
            setDateInput("");
            cancelEditing();

            await Promise.all([
                loadAssignments(nextId),
                loadFrozenDates(nextId),
            ]);

            setMessage(
                `${title} monthly form created successfully.`
            );
        } catch (err) {
            setError(err.message);
        } finally {
            setCreatingMonth(false);
        }
    }

    // --------------------------------------------------
    // SELECTED FORM
    // --------------------------------------------------

    const selectedForm = forms.find(
        (form) => String(form.id) === String(selectedId)
    );

    const isClosed =
        Boolean(selectedForm) && !selectedForm.is_open;

    const selectedFrozenDates = frozenDates;

    function isDateFrozen(dateValue) {
        const date = getDateOnly(dateValue);

        return selectedFrozenDates.some(
            (item) => getDateOnly(item.frozen_date) === date
        );
    }

    function isAssignmentFrozen(item) {
        return isDateFrozen(
            item.assignment_date || item.date
        );
    }

    // --------------------------------------------------
    // ASSIGNMENT COUNTS
    // --------------------------------------------------

    const filled = assignments.filter(
        (item) => item.lector_id != null
    ).length;

    const unfilled = assignments.filter(
        (item) => item.lector_id == null
    ).length;

    const progress =
        assignments.length > 0
            ? Math.round((filled / assignments.length) * 100)
            : 0;

    const calendarGroups = useMemo(
        () => groupCalendarAssignments(assignments),
        [assignments]
    );

    const activeLectors = lectors.filter(
        (lector) => lector.is_active
    );

    // --------------------------------------------------
    // FORM CLOSURE / REOPENING
    // --------------------------------------------------

    async function closeForm() {
        if (!selectedId) return;

        const confirmed = window.confirm(
            "Are you sure you want to close this monthly form?"
        );

        if (!confirmed) return;

        setWorking(true);
        setError("");
        setMessage("");

        try {
            const result = await request(
                `/monthly-forms/${selectedId}/close`,
                {
                    method: "PATCH",
                }
            );

            await loadForms();

            setMessage(
                result.message || "Monthly form closed successfully."
            );
        } catch (err) {
            setError(err.message);
        } finally {
            setWorking(false);
        }
    }

    async function reopenForm() {
        if (!selectedId) return;

        const confirmed = window.confirm(
            "Are you sure you want to reopen this monthly form?"
        );

        if (!confirmed) return;

        setWorking(true);
        setError("");
        setMessage("");

        try {
            const result = await request(
                `/monthly-forms/${selectedId}/reopen`,
                {
                    method: "PATCH",
                }
            );

            await loadForms();

            setMessage(
                result.message || "Monthly form reopened successfully."
            );
        } catch (err) {
            setError(err.message);
        } finally {
            setWorking(false);
        }
    }

    // --------------------------------------------------
    // FROZEN DATES
    // --------------------------------------------------

    async function addFrozenDate() {
        if (!dateInput || !selectedId || !selectedForm) {
            return;
        }

        const date = new Date(`${dateInput}T12:00:00`);

        if (
            date.getMonth() + 1 !== Number(selectedForm.month) ||
            date.getFullYear() !== Number(selectedForm.year)
        ) {
            setError(
                "Choose a date within the selected monthly form."
            );
            return;
        }

        if (isDateFrozen(dateInput)) {
            setError("That date has already been frozen.");
            return;
        }

        setWorking(true);
        setError("");
        setMessage("");

        try {
            await request("/frozen-dates/", {
                method: "POST",
                body: JSON.stringify({
                    monthly_form_id: Number(selectedId),
                    frozen_date: dateInput,
                    reason: null,
                }),
            });

            await loadFrozenDates(selectedId);

            setDateInput("");
            setMessage("Frozen date saved successfully.");
        } catch (err) {
            setError(err.message);
        } finally {
            setWorking(false);
        }
    }

    async function removeFrozenDate(frozenDateId) {
        const confirmed = window.confirm(
            "Are you sure you want to remove this frozen date?"
        );

        if (!confirmed) return;

        setWorking(true);
        setError("");
        setMessage("");

        try {
            await request(`/frozen-dates/${frozenDateId}`, {
                method: "DELETE",
            });

            await loadFrozenDates(selectedId);

            setMessage("Frozen date removed successfully.");
        } catch (err) {
            setError(err.message);
        } finally {
            setWorking(false);
        }
    }

    // --------------------------------------------------
    // GENERATE / AUTO ASSIGN
    // --------------------------------------------------

    async function runAction(action) {
        if (!selectedId) {
            setError("Please select a monthly form first.");
            return;
        }

        if (isClosed) {
            setError(
                "This monthly form is closed. Reopen it before scheduling."
            );
            return;
        }

        const labels = {
            generate: "Generate schedule positions",
            auto: "Run automatic assignments",
        };

        if (
            !window.confirm(
                `${labels[action]} for this monthly form?`
            )
        ) {
            return;
        }

        setWorking(true);
        setError("");
        setMessage("");

        try {
            const path =
                action === "generate"
                    ? `/assignments/generate/${selectedId}`
                    : `/assignments/auto-assign/${selectedId}`;

            const result = await request(path, {
                method: "POST",
            });

            setMessage(
                result.message ||
                (action === "generate"
                    ? "Schedule positions generated."
                    : "Automatic assignments completed.")
            );

            await loadAssignments(selectedId);
        } catch (err) {
            setError(err.message);
        } finally {
            setWorking(false);
        }
    }

    // --------------------------------------------------
    // MANUAL ASSIGNMENT
    // --------------------------------------------------

    function startEditing(item) {
        if (isClosed) {
            setError(
                "Reopen the monthly form before editing assignments."
            );
            return;
        }

        if (isAssignmentFrozen(item)) {
            setError(
                "This date is frozen. Remove the frozen date before editing this assignment."
            );
            return;
        }

        setEditingAssignmentId(item.id);

        setManualLectorId(
            item.lector_id != null
                ? String(item.lector_id)
                : ""
        );

        setOverrideAvailability(false);
        setError("");
        setMessage("");
    }

    function cancelEditing() {
        setEditingAssignmentId(null);
        setManualLectorId("");
        setOverrideAvailability(false);
    }

    async function saveManualAssignment(assignmentId) {
        if (isClosed) {
            setError(
                "Reopen the monthly form before editing assignments."
            );
            return;
        }

        if (!manualLectorId) {
            setError("Please select a lector.");
            return;
        }

        setWorking(true);
        setError("");
        setMessage("");

        try {
            const result = await request(
                `/assignments/${assignmentId}/manual`,
                {
                    method: "PATCH",
                    body: JSON.stringify({
                        lector_id: Number(manualLectorId),
                        override_availability: overrideAvailability,
                    }),
                }
            );

            setMessage(
                result.message ||
                "Manual assignment saved successfully."
            );

            cancelEditing();

            await loadAssignments(selectedId);
        } catch (err) {
            setError(err.message);
        } finally {
            setWorking(false);
        }
    }

    // --------------------------------------------------
    // CLEAR ASSIGNMENT POSITION
    // --------------------------------------------------

    async function clearManualAssignment(assignmentId) {
        if (isClosed) {
            setError(
                "Reopen the monthly form before clearing assignments."
            );
            return;
        }

        const confirmed = window.confirm(
            "This will permanently remove this position from the calendar. Continue?"
        );

        if (!confirmed) return;

        setWorking(true);
        setError("");
        setMessage("");

        try {
            const result = await request(
                `/assignments/${assignmentId}/manual`,
                {
                    method: "DELETE",
                }
            );

            cancelEditing();

            await loadAssignments(selectedId);

            setMessage(
                result.message ||
                "Assignment position removed from the calendar."
            );
        } catch (err) {
            setError(err.message);
        } finally {
            setWorking(false);
        }
    }

    // --------------------------------------------------
    // DOWNLOADS: PDF AND WORD ONLY
    // --------------------------------------------------

    function downloadRoster(format) {
        if (!selectedForm || assignments.length === 0) {
            setError(
                "Load assignments before downloading the Lectors List."
            );
            return;
        }

        const html = buildRosterDocument(
            selectedForm,
            assignments
        );

        const monthLabel =
            selectedForm.title ||
            `${selectedForm.month || "month"}-${selectedForm.year || ""}`;

        const safeName = String(monthLabel)
            .trim()
            .replace(/[^a-z0-9-_]+/gi, "-");

        if (format === "word") {
            const blob = new Blob(["\ufeff", html], {
                type: "application/msword;charset=utf-8",
            });

            const url = URL.createObjectURL(blob);
            const link = document.createElement("a");

            link.href = url;
            link.download = `Lectors-List-${safeName}.doc`;

            document.body.appendChild(link);
            link.click();
            link.remove();

            URL.revokeObjectURL(url);
            return;
        }

        if (format === "pdf") {
            const printWindow = window.open("", "_blank");

            if (!printWindow) {
                setError(
                    "Allow pop-ups for this site to create the PDF."
                );
                return;
            }

            printWindow.document.open();
            printWindow.document.write(html);
            printWindow.document.close();

            printWindow.onload = () => {
                printWindow.focus();
                printWindow.print();
            };
        }
    }
